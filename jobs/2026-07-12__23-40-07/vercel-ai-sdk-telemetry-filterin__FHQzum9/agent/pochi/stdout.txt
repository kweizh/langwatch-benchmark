  process.env.LANGWATCH_ENDPOINT ?? "http://127.0.0.1:9999";

  process.env.LANGWATCH_API_KEY ?? "dummy-api-key-for-offline-run";

// ---------------------------------------------------------------------------
// CLI argument parsing
// ---------------------------------------------------------------------------

function parsePromptArg(): string {
  const argv = process.argv.slice(2);
  const idx = argv.findIndex(
    (a) => a === "--prompt" || a.startsWith("--prompt="),
  );
  if (idx === -1) {
    throw new Error("Missing required --prompt argument");
  }
  const raw = argv[idx];
  if (raw.startsWith("--prompt=")) {
    return raw.slice("--prompt=".length);
  }
  const value = argv[idx + 1];
  if (value === undefined) {
    throw new Error("--prompt was provided without a value");
  }
  return value;
}

// ---------------------------------------------------------------------------
// Client-side payload filtering
// ---------------------------------------------------------------------------

const SENSITIVE_TOKEN = "SECRET_TOKEN";
const REDACTION_PLACEHOLDER = "[REDACTED]";

/**
 * Recursively walks a value and replaces every occurrence of `SECRET_TOKEN`
 * with `[REDACTED]` inside any string we find. Objects and arrays are
 * traversed, dates and other primitives are returned as-is.
 */
function redactValue(value: unknown): unknown {
  if (typeof value === "string") {
    return value.split(SENSITIVE_TOKEN).join(REDACTION_PLACEHOLDER);
  }
  if (Array.isArray(value)) {
    return value.map((v) => redactValue(v));
  }
  if (value && typeof value === "object") {
    const out: Record<string, unknown> = {};
    for (const [k, v] of Object.entries(value as Record<string, unknown>)) {
      out[k] = redactValue(v);
    }
    return out;
  }
  return value;
}

/**
 * Returns a sanitized copy of a span's attributes with all `SECRET_TOKEN`
 * occurrences replaced by `[REDACTED]`. The original span is never mutated.
 */
function redactSpanAttributes(
  span: ReadableSpan,
): Record<string, unknown> {
  return redactValue(span.attributes ?? {}) as Record<string, unknown>;
}

/**
 * Mutates a span so that its `attributes` field reflects the sanitized
 * values. The OTel SDK's `Span` class keeps a mutable `attributes` object
 * even though the public type marks it as readonly. Mutating it in-place
 * guarantees that every downstream serializer (OTLP transformer, etc.)
 * sees the redacted values without us having to reimplement the protobuf
 * encoding.
 */
function applyRedactionToSpan(span: ReadableSpan): ReadableSpan {
  const sanitized = redactSpanAttributes(span);
  const anySpan = span as unknown as {
    attributes?: Record<string, unknown>;
  };
  if (anySpan.attributes && typeof anySpan.attributes === "object") {
    for (const key of Object.keys(anySpan.attributes)) {
      delete anySpan.attributes[key];
    }
    Object.assign(anySpan.attributes, sanitized);
  }
  return span;
}

/**
 * Wraps a `SpanExporter` and applies a `dataCapture` style transform to every
 * span before it is forwarded to the inner exporter. This is the
 * LangWatch-compatible equivalent of the `dataCapture` callback used to
 * redact sensitive information before telemetry leaves the client.
 */
class FilteringSpanExporter {
  private readonly inner: {
    export: (s: ReadableSpan[], cb: (r: any) => void) => void;
    shutdown?: () => Promise<void>;
  };
  public readonly dataCapture: (spans: ReadableSpan[]) => ReadableSpan[];
  public exportedPayloads: { serialized: string; spans: ReadableSpan[] }[] = [];

  constructor(params: {
    inner: {
      export: (s: ReadableSpan[], cb: (r: any) => void) => void;
      shutdown?: () => Promise<void>;
    };
    dataCapture?: (spans: ReadableSpan[]) => ReadableSpan[];
  }) {
    this.inner = params.inner;
    this.dataCapture =
      params.dataCapture ??
      ((spans) => spans);
  }

  export(spans: ReadableSpan[], resultCallback: (result: any) => void): void {
    const filtered = this.dataCapture(spans).map(applyRedactionToSpan);

    // Cache the JSON payload for inspection / testing. We don't need to
    // re-serialize in production, but it makes it trivial to verify the
    // redaction in tests and from the CLI.
    try {
      const serialized = JSON.stringify(
        filtered.map((s) => ({
          name: s.name,
          attributes: s.attributes,
        })),
      );
      this.exportedPayloads.push({ serialized, spans: filtered });
    } catch {
      // ignore serialization errors
    }

    this.inner.export(filtered, resultCallback);
  }

  async shutdown(): Promise<void> {
    if (this.inner.shutdown) {
      await this.inner.shutdown();
    }
  }

  async forceFlush(): Promise<void> {
    // Nothing buffered here; flushing is delegated to the tracer provider.
  }
}

/**
 * Initializes OpenTelemetry-based LangWatch observability for Node.js.
 *
 * Returns an object exposing:
 *  - `provider`: the tracer provider used for AI SDK telemetry
 *  - `tracer`: a tracer the AI SDK can use via `experimental_telemetry.tracer`
 *  - `filteringExporter`: the wrapper exporter that redacts sensitive tokens
 *  - `memoryExporter`: an in-memory exporter that captures spans locally so we
 *    can assert that no `SECRET_TOKEN` leaks into outgoing payloads
 *  - `shutdown()`: flushes and shuts the provider down
 */
function setupObservability(): {
  provider: BasicTracerProvider;
  tracer: Tracer;
  filteringExporter: FilteringSpanExporter;
  memoryExporter: InMemorySpanExporter;
  shutdown: () => Promise<void>;
} {
  const langwatchExporter = new LangWatchExporter({
    endpoint: LANGWATCH_ENDPOINT,
    apiKey: LANGWATCH_API_KEY,
    debug: false,
  });

  const memoryExporter = new InMemorySpanExporter();

  // The dataCapture callback receives every span that is about to be exported
  // and returns the (possibly modified) spans that should actually be sent.
  // This is exactly the filtering mechanism documented for LangWatch's
  // payload redaction, applied here directly to the OTEL exporter pipeline.
  const dataCapture = (spans: ReadableSpan[]): ReadableSpan[] => spans;

  const filteringExporter = new FilteringSpanExporter({
    inner: langwatchExporter,
    dataCapture,
  });

  const provider = new BasicTracerProvider();
  // SimpleSpanProcessor exports each span as soon as it ends, which makes it
  // trivial to flush all telemetry before the script exits without having to
  // wait for a batch interval.
  provider.addSpanProcessor(
    new SimpleSpanProcessor(filteringExporter as any),
  );
  // Also feed spans to the in-memory exporter so we can verify the filter
  // applied correctly without needing to intercept the HTTP traffic.
  provider.addSpanProcessor(new SimpleSpanProcessor(memoryExporter));

  provider.register();

  const tracer = trace.getTracer("langwatch-vercel-ai-sdk-demo");

  return {
    provider,
    tracer,
    filteringExporter,
    memoryExporter,
    shutdown: async () => {
      await provider.forceFlush();
      await provider.shutdown();
    },
  };
}

/**
 * Extract the user's text from the prompt argument. The Vercel AI SDK accepts
 * either a string or an array of `LanguageModelV1Message` objects depending
 * on the entry point; we handle both for completeness.
 */
function extractPromptText(prompt: unknown): string {
  if (typeof prompt === "string") return prompt;
  if (Array.isArray(prompt)) {
    return prompt
      .map((part: any) => {
        if (typeof part === "string") return part;
        if (part && typeof part === "object" && "content" in part) {
          const content = (part as any).content;
          if (typeof content === "string") return content;
          if (Array.isArray(content)) {
            return content
              .map((c: any) => (typeof c === "string" ? c : c?.text ?? ""))
              .join(" ");
          }
        }
        if (part && typeof part === "object" && "text" in part) {
          return String((part as any).text ?? "");
        }
        return "";
      })
      .join(" ");
  }
  return "";
}

/**
 * A minimal `LanguageModelV1` implementation that echoes the prompt back as
 * the model's response. It exists so we can exercise the full telemetry
 * pipeline (and the redaction logic) without any real LLM API key.
 */
function createMockLanguageModel(): LanguageModelV1 {
  return {
    specificationVersion: "v1",
    provider: "mock-provider",
    modelId: "mock-echo-model",
    defaultObjectGenerationMode: undefined,

    async doGenerate(options: any) {
      const promptText = extractPromptText(options?.prompt);
      const reply = `Mock response to: ${promptText}`;
      return {
        text: reply,
        finishReason: "stop",
        usage: {
          promptTokens: promptText.length,
          completionTokens: reply.length,
        },
        rawCall: {
          rawPrompt: options?.prompt ?? null,
          rawSettings: {},
        },
        warnings: [],
      };
    },

    async doStream(options: any) {
      const promptText = extractPromptText(options?.prompt);
      const reply = `Mock response to: ${promptText}`;
      const stream = new ReadableStream<any>({
        start(controller) {
          controller.enqueue({ type: "response-metadata", id: "mock" });
          controller.enqueue({ type: "text-delta", textDelta: reply });
          controller.enqueue({
            type: "finish",
            finishReason: "stop",
            usage: {
              promptTokens: promptText.length,
              completionTokens: reply.length,
            },
          });
          controller.close();
        },
      });
      return {
        stream,
        rawCall: {
          rawPrompt: options?.prompt ?? null,
          rawSettings: {},
        },
      };
    },
  };
}

async function main() {
  const prompt = parsePromptArg();

  const observability = setupObservability();

  try {
    const result = await generateText({
      model: createMockLanguageModel() as any,
      prompt,
      experimental_telemetry: {
        isEnabled: true,
        tracer: observability.tracer,
      },
    });

    // Print the generated text as required by the acceptance criteria.
    process.stdout.write(`${result.text}\n`);

    // Allow the in-flight span to be exported.
    await observability.shutdown();

    // Verify the redaction worked on the in-memory copy as well.
    const leaked = verifyNoSensitiveLeak(observability.memoryExporter);
    if (leaked.length > 0) {
      console.error(
        `\n[run.ts] ERROR: SECRET_TOKEN leaked into exported telemetry on ${leaked.length} span(s):`,
      );
      for (const name of leaked) {
        console.error(`  - ${name}`);
      }
      process.exitCode = 1;
    } else {
      console.error(
        "\n[run.ts] Verified: no SECRET_TOKEN strings in exported telemetry payloads.",
      );
    }
  } catch (err) {
    await observability.shutdown().catch(() => {});
    throw err;
  }
}

function verifyNoSensitiveLeak(exporter: InMemorySpanExporter): string[] {
  const offending: string[] = [];
  for (const span of exporter.getFinishedSpans()) {
    const serialized = JSON.stringify(span.attributes ?? {});
    if (serialized.includes(SENSITIVE_TOKEN)) {
      offending.push(span.name);
    }
  }
  return offending;
}

main().catch((err) => {
  console.error("[run.ts] Failed:", err);
  process.exit(1);
});

// Verification: asserts that spans generated by generateText contain
// SECRET_TOKEN in their attributes BEFORE the filter is applied and
// contain [REDACTED] (with no SECRET_TOKEN) AFTER the filter is applied.

import { generateText } from "ai";
import { LangWatchExporter } from "langwatch";
import {
  BasicTracerProvider,
  SimpleSpanProcessor,
  ReadableSpan,
} from "@opentelemetry/sdk-trace-base";
import { trace } from "@opentelemetry/api";
import * as fs from "fs";

process.env.LANGWATCH_ENDPOINT ??= "http://127.0.0.1:9999";
process.env.LANGWATCH_API_KEY ??= "dummy-api-key-for-offline-run";

const SENSITIVE_TOKEN = "SECRET_TOKEN";
const REDACTION_PLACEHOLDER = "[REDACTED]";

function redactValue(value: unknown): unknown {
  if (typeof value === "string") {
    return value.split(SENSITIVE_TOKEN).join(REDACTION_PLACEHOLDER);
  }
  if (Array.isArray(value)) {
    return value.map((v) => redactValue(v));
  }
  if (value && typeof value === "object") {
    const out: Record<string, unknown> = {};
    for (const [k, v] of Object.entries(value as Record<string, unknown>)) {
      out[k] = redactValue(v);
    }
    return out;
  }
  return value;
}

function applyRedactionToSpan(span: ReadableSpan): ReadableSpan {
  const sanitized = redactValue(span.attributes ?? {}) as Record<string, unknown>;
  const anySpan = span as unknown as {
    attributes?: Record<string, unknown>;
  };
  if (anySpan.attributes && typeof anySpan.attributes === "object") {
    for (const key of Object.keys(anySpan.attributes)) {
      delete anySpan.attributes[key];
    }
    Object.assign(anySpan.attributes, sanitized);
  }
  return span;
}

// A wrapper exporter that captures BOTH the unfiltered (raw) payload and
// the filtered (redacted) payload, plus what it would forward to the inner
// exporter (the LangWatchExporter).
class RecordingExporter {
  public rawPayloads: string[] = [];
  public forwardedPayloads: string[] = [];

  constructor(private inner: any) {}

  export(spans: ReadableSpan[], resultCallback: any): void {
    // Capture raw (before filtering)
    this.rawPayloads.push(
      JSON.stringify(spans.map((s) => ({ name: s.name, attributes: s.attributes }))),
    );

    // Apply filter (mutates spans in place)
    const filtered = spans.map(applyRedactionToSpan);

    // Capture what would be forwarded to LangWatch
    this.forwardedPayloads.push(
      JSON.stringify(filtered.map((s) => ({ name: s.name, attributes: s.attributes }))),
    );

    // Do not call inner.export to avoid network errors during the test
    resultCallback({ code: 0 });
  }

  async shutdown(): Promise<void> {}
  async forceFlush(): Promise<void> {}
}

    defaultObjectGenerationMode: undefined,
    async doGenerate(options: any) {

      const reply = `Mock response to: ${promptText}`;
      return {
        text: reply,
        finishReason: "stop",
        usage: { promptTokens: promptText.length, completionTokens: reply.length },
        rawCall: { rawPrompt: options?.prompt ?? null, rawSettings: {} },
        warnings: [],
      };
    },
    async doStream() {
      throw new Error("not used");
    },
  };
}

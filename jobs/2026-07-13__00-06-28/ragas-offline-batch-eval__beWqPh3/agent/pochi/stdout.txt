from difflib import SequenceMatcher

def retrieve(question, kb, top_k=3):
    scored = []
    ql = question.lower()
    for doc in kb:
        content = doc['content']
        score = SequenceMatcher(None, ql, content.lower()).ratio()
        scored.append((score, content))
    scored.sort(key=lambda x: x[0], reverse=True)
    return [c for _, c in scored[:top_k]]

def generate(question, contexts):
    if not contexts:
        return 'No relevant information found.'
    ql = question.lower()
    best_sentence = ''
    best_score = -1.0
    for ctx in contexts:
        sentences = ctx.replace('\n', ' ').split('. ')
        for s in sentences:
            s = s.strip()
            if not s:
                continue
            score = SequenceMatcher(None, ql, s.lower()).ratio()
            if score > best_score:
                best_score = score
                best_sentence = s
    return best_sentence or contexts[0]

for r in rows:
    ctxs = retrieve(r['question'], kb)
    out = generate(r['question'], ctxs)
    print(f'Q: {r[\"question\"]}')
    print(f'  Contexts: {len(ctxs)} docs')
    print(f'  Output: {out[:80]}...')
    print(f'  Output non-empty: {bool(out)}')
    print(f'  Contexts non-empty: {bool(ctxs)}')
    print()

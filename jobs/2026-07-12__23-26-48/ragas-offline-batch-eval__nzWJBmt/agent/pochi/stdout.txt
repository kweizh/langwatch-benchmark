def clean_and_stem(text):

    cleaned = []
    for w in words:
        if w in stopwords:
            continue
        stem = w
        if len(stem) > 4:

# Calculate Document Frequency (DF) for each stem in the knowledge base
df_counts = collections.Counter()
for doc in kb:
    stems = set(clean_and_stem(doc['title']) + clean_and_stem(doc['content']))
    for stem in stems:
        df_counts[stem] += 1

# Total number of documents
N = len(kb)

def retrieve_tfidf(question, kb):
    qw = clean_and_stem(question)
    best_doc = None
    best_score = -1
    for doc in kb:
        doc_stems = clean_and_stem(doc['title']) + clean_and_stem(doc['content'])
        score = 0
        for stem in qw:
            if stem in doc_stems:
                tf = doc_stems.count(stem)
                idf = math.log((N + 1) / (df_counts[stem] + 1)) + 1
                score += tf * idf
        if score > best_score:
            best_score = score
            best_doc = doc
    return best_doc['content'] if best_doc else kb[0]['content']

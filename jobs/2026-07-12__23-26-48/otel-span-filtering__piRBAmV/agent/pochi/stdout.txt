import os

import langwatch
from langwatch.domain import SpanProcessingExcludeRule

provider = TracerProvider()
provider.add_span_processor(SimpleSpanProcessor(ConsoleSpanExporter()))

exclude_rules = [
    SpanProcessingExcludeRule(

        match_value='db.',
        match_operation='starts_with'
